class solution:
    def checkArmstrongNumber(self, num):
        #Write your code here...
        org = num
        digits = 0 
        temp = num
        while temp > 0:
            digits += 1
            temp //= 10
        total = 0
        temp = num
        while temp > 0:
            digit = temp % 10
            total += digit ** digits
            temp //= 10
        if total == org:
            return True
        else:
            return False
